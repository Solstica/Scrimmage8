\begin{algorithm}[!htbp]
\caption{问题一微小时间段分离变量求解流程}
\label{alg:q1-short-time-separation}
\begin{algorithmic}[1]
\Require 三层厚度与热物性，初始温度$37\,^{\circ}\mathrm C$，环境温度$-40\,^{\circ}\mathrm C$，附件放热曲线$q_{\rm pcm}(T)$，时间步$\Delta t$
\Ensure 温度场$T(x,t)$、人体侧温度$T_s(t)$及阈值时间$t_{15},t_{10}$
\State 对附件放热数据建立保形插值函数$q_{\rm pcm}(T)$
\State 初始化$T_i(x,0)=37\,^{\circ}\mathrm C$，令$n\gets0$
\While{$T_s(t_n)>10\,^{\circ}\mathrm C$}
    \State 由当前功能层温度计算$q_{\rm pcm}(\bar T_2^n)$，并在$[t_n,t_{n+1}]$内冻结该热源
    \State 将两个界面温度写成局部线性形式，未知斜率记为$\beta_{12}^n,\beta_{23}^n$
    \State 对三层分别进行边界齐次化，并由分离变量法得到该时间段的解析温度表达式
    \State 代入两个界面的热流连续条件，求$\beta_{12}^n,\beta_{23}^n$
    \State 更新$t_{n+1}$时刻的三层温度场，并计算$T_s(t_{n+1})$
    \State 若首次越过$15\,^{\circ}\mathrm C$或$10\,^{\circ}\mathrm C$，则在相邻时间点间插值确定$t_{15}$或$t_{10}$
    \State $n\gets n+1$
\EndWhile
\State 以$\Delta t/2$重复计算，检查$t_{15}$和$t_{10}$的时间步收敛性
\end{algorithmic}
\end{algorithm}
