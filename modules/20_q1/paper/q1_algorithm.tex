\begin{algorithm}[!htbp]
\caption{恒温人体边界下的微小时间段分离变量求解流程}
\label{alg:q1-short-time-separation}
\begin{algorithmic}[1]
\Require 三层厚度与热物性，人体恒温$T_b=37\,^{\circ}\mathrm C$，人体侧$h_b=3.0\,\mathrm{W/(m^2\cdot K)}$，环境温度$T_\infty=-40\,^{\circ}\mathrm C$，附件DSC曲线$p(T)$，相变总潜热$L_{\rm pcm}$，时间步$\Delta t$，模态截断数$M$
\Ensure 三层温度场$T_i(x,t)$、贴身侧温度$T_{\rm in}(t)=T_1(0,t)$及阈值时间$t_{15},t_{10}$
\State 对附件DSC曲线完成基线、符号和单位整理，计算归一化累计面积得到$\xi_{\rm DSC}(T)$及$-\mathrm d\xi_{\rm DSC}/\mathrm dT$
\State 初始化$T_i(x,0)=37\,^{\circ}\mathrm C$，相变进度$\xi(x,0)=0$，令$n\gets0$
\While{$T_{\rm in}(t_n)>10\,^{\circ}\mathrm C$}
    \State 由当前功能层温度计算$\bar T_2^n$，更新$c_{2,\rm eff}^n=c_2-L_{\rm pcm}\xi'_{\rm DSC}(\bar T_2^n)$与$\alpha_2^n=k_2/(\rho_2c_{2,\rm eff}^n)$
    \State 对相变进度施加历史约束$\xi^{n+1}=\max\{\xi^n,\xi_{\rm DSC}(T_2)\}$，禁止已固化部分因局部回温而反向减少
    \State 由当前服装外表面温度计算$h_e^n=2.38[T_3(L,t_n)-T_\infty]^{0.25}$，在当前时间段内冻结
    \State 令$T_{12},T_{23}$在当前微小时间段内线性变化，未知斜率为$\beta_{12}^n,\beta_{23}^n$；人体边界$T_b$保持常数
    \State 第一层采用Robin--Dirichlet本征展开，第二层采用Dirichlet--Dirichlet正弦展开，第三层采用Dirichlet--Robin本征展开
    \State 由两个材料界面的段末热流连续条件联立求$\beta_{12}^n,\beta_{23}^n$
    \State 代入各层模态解析式，更新$t_{n+1}$时刻的三层温度场及$T_{\rm in}(t_{n+1})$
    \State 若$T_{\rm in}$首次越过$15\,^{\circ}\mathrm C$或$10\,^{\circ}\mathrm C$，则在相邻时间点间插值确定$t_{15}$或$t_{10}$
    \State $n\gets n+1$
\EndWhile
\State 以$\Delta t/2$和更大的$M$重复计算，检查$t_{15},t_{10}$的时间步与模态截断收敛性，并与隐式有限体积基线核对
\end{algorithmic}
\end{algorithm}
