\begin{algorithm}[!htbp]
\caption{Micro-interval separated-variable solution under a fixed body-temperature boundary}
\label{alg:q1-short-time-separation}
\KwIn{layer thickness and thermophysical properties, fixed body temperature $T_b=37\,^{\circ}\mathrm C$, body-side convection $h_b=3.0\,\mathrm{W/(m^2\cdot K)}$, ambient temperature $T_\infty=-40\,^{\circ}\mathrm C$, baseline-corrected DSC curve $p(T)$, scan rate $\beta_{\rm DSC}$, time step $\Delta t$, modal truncation $M$}
\KwOut{three-layer temperature fields $T_i(x,t)$, body-side temperature $T_{\rm in}(t)=T_1(0,t)$, threshold times $t_{15},t_{10}$}
Apply the endpoint-linear operational baseline to the DSC data and compute $A_{\rm DSC}=\int p(T)\,\mathrm dT$\;
Set $L_{\rm pcm}=60A_{\rm DSC}/\beta_{\rm DSC}$, with baseline $\beta_{\rm DSC}=5\ \mathrm{K/min}$\;
Build the normalized cumulative progress $\xi_{\rm DSC}(T)$ and its derivative $-\mathrm d\xi_{\rm DSC}/\mathrm dT$\;
Initialize $T_i(x,0)=37\,^{\circ}\mathrm C$, solidification progress $\xi(x,0)=0$, and set $n\gets0$\;
\While{$T_{\rm in}(t_n)>10\,^{\circ}\mathrm C$}{
  Evaluate the layer-mean temperature $\bar T_2^n$ and the candidate progress $\xi_{\rm trial}=\xi_{\rm DSC}(\bar T_2^n)$\;
  \eIf{$\xi_{\rm trial}>\xi^n$}{
    Advance solidification: set $c_{2,\rm eff}^n=c_2-L_{\rm pcm}\xi'_{\rm DSC}(\bar T_2^n)$ and $\xi^{n+1}=\xi_{\rm trial}$\;
  }{
    Freeze solidification: set $c_{2,\rm eff}^n=c_2$ and $\xi^{n+1}=\xi^n$\;
  }
  Compute $\alpha_2^n=k_2/(\rho_2c_{2,\rm eff}^n)$, where the history constraint keeps $\xi$ nondecreasing\;
  Update $h_e^n=2.38[T_3(L,t_n)-T_\infty]^{0.25}$ and freeze it on $[t_n,t_{n+1}]$\;
  Approximate $T_{12},T_{23}$ as linear in $\tau$ with unknown slopes $\beta_{12}^n,\beta_{23}^n$ while $T_b$ stays constant\;
  Expand layer 1 with Robin--Dirichlet eigenfunctions, layer 2 with Dirichlet--Dirichlet sines, and layer 3 with Dirichlet--Robin eigenfunctions\;
  Enforce heat-flux continuity at both material interfaces at $\tau=\Delta t$ to close $\beta_{12}^n,\beta_{23}^n$\;
  Advance the full temperature field to $t_{n+1}$ by the modal solutions and update $T_{\rm in}(t_{n+1})$\;
  If $T_{\rm in}$ first crosses $15\,^{\circ}\mathrm C$ or $10\,^{\circ}\mathrm C$, interpolate $t_{15}$ or $t_{10}$ between adjacent time nodes\;
  $n\gets n+1$\;
}
\Return{$T_i(x,t)$, $T_{\rm in}(t)$, $t_{15}$, $t_{10}$}\;
\end{algorithm}
