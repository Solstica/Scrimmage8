\subsection{预备工作}

\noindent\textbf{1. 几何与材料参数}\par

三层厚度分别记为$d_1,d_2,d_3$，层间位置为$x_1=d_1$、$x_2=d_1+d_2$，总厚度$L=d_1+d_2+d_3$。附件给出$d_1=0.7\ \mathrm{mm}$、$d_2=0.4\ \mathrm{mm}$、$d_3=0.3\ \mathrm{mm}$，三层热物性分别记为$\rho_i,c_i,k_i$。近期多层防护服研究亦将多层织物间的热量传递作为核心建模对象\cite{yang2025multilayer}。

人体—防护服—环境的传热结构与坐标系见图\ref{fig:q1-schematic}，热流由恒温人体依次穿过三层衣料流向低温环境，贴身侧界面温度$T_{\rm in}=T_1(0,t)$作为后续评价量。

\subsection{基于等效热容法的三层非稳态导热模型}

\noindent\textbf{1. 控制方程与相变等效热容}\par

织物层和隔热层满足一维非稳态导热方程
\begin{equation}
\rho_i c_i\frac{\partial T_i}{\partial t}=k_i\frac{\partial^2T_i}{\partial x^2},\qquad i=1,3.
\label{eq:q1-governing-13}
\end{equation}

功能层以不可逆固化进度$\xi$描述有限潜热释放。设固化开始、结束温度分别为$T_s,T_f$，由校正后放热峰定义
\begin{equation}
\xi_{\rm DSC}(T)=
\begin{cases}
0,&T\ge T_s,\\[1mm]
\displaystyle\frac{\int_T^{T_s}p(\theta)\,\mathrm d\theta}{\int_{T_f}^{T_s}p(\theta)\,\mathrm d\theta},&T_f<T<T_s,\\[4mm]
1,&T\le T_f,
\end{cases}
\qquad
\xi(x,t)=\max_{0\le\tau\le t}\xi_{\rm DSC}[T_2(x,\tau)].
\label{eq:q1-xi-dsc}
\end{equation}
历史最大值约束保证固化过程不可逆。记功能层单位质量总固化潜热为$L_{\rm pcm}$，则相变释放满足$Q_{\rm lat}=\rho_2L_{\rm pcm}\,\partial\xi/\partial t$。在单调降温的主要阶段，可将功能层写成等效热容形式
\begin{equation}
\rho_2c_{2,\rm eff}(T_2)\frac{\partial T_2}{\partial t}=k_2\frac{\partial^2T_2}{\partial x^2},
\qquad
c_{2,\rm eff}(T)=c_2+L_{\rm pcm}\frac{p(T)}{\int_{T_f}^{T_s}p(\theta)\,\mathrm d\theta}.
\label{eq:q1-ceff}
\end{equation}
固化完成后等效比热退回材料显热比热$c_2$。附件未给出扫描速率，本文取$\beta_{\rm DSC}=5\ \mathrm{K/min}$为基准，并由$L_{\rm pcm}=60A_{\rm DSC}/\beta_{\rm DSC}$得到$L_{\rm pcm}=244.63\ \mathrm{kJ/kg}$；同时设置$2$和$10\ \mathrm{K/min}$两组敏感性工况。

\noindent\textbf{2. 初始、界面与边界条件}\par

进入低温环境瞬间，三层衣料统一取$T_i(x,0)=37\,^{\circ}\mathrm C$。忽略层间接触热阻，在$x=x_1,x_2$处满足温度与热流连续：
\begin{equation}
\left\{
\begin{aligned}
&T_1(x_1^-,t)=T_2(x_1^+,t),\quad k_1T_{1,x}(x_1^-,t)=k_2T_{2,x}(x_1^+,t),\\
&T_2(x_2^-,t)=T_3(x_2^+,t),\quad k_2T_{2,x}(x_2^-,t)=k_3T_{3,x}(x_2^+,t).
\end{aligned}
\right.
\label{eq:q1-interface}
\end{equation}
人体侧取$h_b=3.0\ \mathrm{W/(m^2\cdot K)}$，外侧自然对流系数按附件关系随衣表温度更新：
\begin{equation}
-k_1T_{1,x}(0,t)=h_b[T_b-T_1(0,t)],
\qquad
-k_3T_{3,x}(L,t)=2.38[T_3(L,t)+40]^{1.25}.
\label{eq:q1-boundaries}
\end{equation}
定义$T_{\rm in}(t)=T_1(0,t)$，并以其首次达到$15\,^{\circ}\mathrm C$和$10\,^{\circ}\mathrm C$的时刻作为评价指标。


\noindent\textbf{3. 模型汇总}\par
综上，我们建立基于等效热容法的三层非稳态导热模型。
\vspace{-0.3\baselineskip}
{\boldmath
\begin{equation}
\left\{
\begin{aligned}
&\rho_i c_iT_{i,t}=k_iT_{i,xx},\quad i=1,3,\\
&\rho_2c_{2,\rm eff}(T_2)T_{2,t}=k_2T_{2,xx},\quad c_{2,\rm eff}(T)=c_2+L_{\rm pcm}\frac{p(T)}{\int_{T_f}^{T_s}p(\theta)\,\mathrm d\theta},\\
&T_i(x,0)=37\,^{\circ}\mathrm C,\quad i=1,2,3,\\
&T_1(x_1^-)=T_2(x_1^+),\quad k_1T_{1,x}(x_1^-)=k_2T_{2,x}(x_1^+),\\
&T_2(x_2^-)=T_3(x_2^+),\quad k_2T_{2,x}(x_2^-)=k_3T_{3,x}(x_2^+),\\
&-k_1T_{1,x}(0,t)=h_b[T_b-T_1(0,t)],\\
&-k_3T_{3,x}(L,t)=2.38[T_3(L,t)+40]^{1.25},\\
&T_{\rm in}(t)=T_1(0,t),\quad t_{15,10}=\inf\{t>0:T_{\rm in}(t)\le15,10\,^{\circ}\mathrm C\}.
\end{aligned}
\right.
\label{eq:q1-model-summary}
\end{equation}}

\subsection{模型求解}

\noindent\textbf{1. 微小时间段参数冻结}\par

为求解多层瞬态导热问题，本文采用分段参数冻结与分层本征展开相结合的拓展分离变量框架。近期多层瞬态导热研究表明，分离变量与本征展开方法可有效处理层合介质的瞬态温度场\cite{jain2024multilayer}。将时间轴划分为$t_n\le t\le t_{n+1}$，局部时间$\tau=t-t_n$。每个微时间段内将两处界面温度近似为
\begin{equation}
T_{12}(\tau)\approx T_{12}^n+\beta_{12}^n\tau,\qquad
T_{23}(\tau)\approx T_{23}^n+\beta_{23}^n\tau.
\label{eq:q1-interface-linear}
\end{equation}
以功能层起点平均温度$\bar T_2^n=d_2^{-1}\int_{x_1}^{x_2}T_2(x,t_n)\,\mathrm dx$表征当前相变状态，并在该时间段内冻结
\begin{equation}
c_{2,\rm eff}^n=c_{2,\rm eff}(\bar T_2^n),\qquad
\alpha_2^n=\frac{k_2}{\rho_2c_{2,\rm eff}^n},\qquad
h_e^n=2.38[T_3(L,t_n)-T_\infty]^{0.25}.
\label{eq:q1-ceff-freeze}
\end{equation}

\noindent\textbf{2. 分层本征展开与界面闭合}\par

对各层先构造满足非齐次边界的线性函数，再对剩余齐次问题进行本征展开。第一层为Robin--Dirichlet边界，第二层为Dirichlet--Dirichlet边界，第三层为Dirichlet--Robin边界，其本征值分别满足
\begin{equation}
\tan(\lambda_{1,m}d_1)=-\frac{k_1\lambda_{1,m}}{h_b},\qquad
\lambda_{2,m}=\frac{m\pi}{d_2},\qquad
\tan(\lambda_{3,m}d_3)=-\frac{k_3\lambda_{3,m}}{h_e^n}.
\label{eq:q1-eigen-summary}
\end{equation}
三层模态系数统一满足
\begin{equation}
B'_{i,m}+\gamma_{i,m}B_{i,m}=F_{i,m}^n,\qquad
B_{i,m}(\tau)=B_{i,m}(0)e^{-\gamma_{i,m}\tau}+\frac{F_{i,m}^n}{\gamma_{i,m}}\left(1-e^{-\gamma_{i,m}\tau}\right),
\label{eq:q1-mode-solution}
\end{equation}
其中$\gamma_{i,m}=\alpha_i\lambda_{i,m}^2$，第二层取$\alpha_2=\alpha_2^n$。在时间段末端以两处界面热流连续条件
\begin{equation}
k_1T_{1,x}(x_1,t)=k_2T_{2,x}(x_1,t),\qquad
k_2T_{2,x}(x_2,t)=k_3T_{3,x}(x_2,t)
\end{equation}
联立求得$\beta_{12}^n,\beta_{23}^n$，再由模态解析式推进至$t_{n+1}$。完整计算流程见算法\ref{alg:q1-thermal}。

\subsection{数值收敛与复算检验}

基准工况下时间步减半后$t_{15},t_{10}$分别变化$0.0019\%$和$0.0010\%$，模态数增至$60$后变化$0.291\%$和$0.283\%$。独立隐式有限体积法复算三组工况，与模态法的阈值时间偏差为$0.81\%$--$0.99\%$。数值离散影响明显小于扫描速率引起的跨工况差异，当前设置具有良好的数值稳定性与复算一致性。
