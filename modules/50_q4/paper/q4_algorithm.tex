\begin{algorithm}[H]
\SetAlgoLined
\setstretch{0.92}
\SetAlFnt{\normalsize}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{Input}
\SetKwInput{KwOut}{Output}
\SetKwInput{KwInit}{Initialize}
\SetKwFor{While}{while}{do}{end while}
\SetKwIF{If}{ElseIf}{Else}{if}{then}{else if}{else}{end if}
\SetKw{KwRet}{Return}
\caption{问题四最小放热倍率二分搜索}
\label{alg:q4-bisection}
\KwIn{$\beta_{\rm DSC},\theta,t_{3,\theta}^*$，$t_W^{(4)}$，$\varepsilon_\alpha$，$L_{\rm pcm}$}
\KwInit{$\alpha_L\leftarrow1$，$\alpha_U\leftarrow2$；定义$F(\alpha)=\min\{t_{T,\theta}^{(4)}(\alpha),t_W^{(4)}\}$}
\KwOut{$\alpha^*$、$r_{\min}=100(\alpha^*-1)\%$及$F(\alpha^*)$}
\If{$t_W^{(4)}<t_{3,\theta}^*$}{\KwRet{无可行解}\;}
计算$F(1)$；\If{$F(1)\ge t_{3,\theta}^*$}{\KwRet{$\alpha^*=1$，$r_{\min}=0$，$F(1)$}\;}
计算$F(\alpha_U)$\;
\While{$F(\alpha_U)<t_{3,\theta}^*$}{
  $\alpha_L\leftarrow\alpha_U$，$\alpha_U\leftarrow2\alpha_U$；计算$F(\alpha_U)$\;
}
\While{$\alpha_U-\alpha_L>\varepsilon_\alpha$}{
  $\alpha_M\leftarrow(\alpha_L+\alpha_U)/2$；计算$F(\alpha_M)$\;
  \eIf{$F(\alpha_M)\ge t_{3,\theta}^*$}{$\alpha_U\leftarrow\alpha_M$\;}{$\alpha_L\leftarrow\alpha_M$\;}
}
$\alpha^*\leftarrow\alpha_U$；复算$F(\alpha^*)$\;
\KwRet{$\alpha^*$，$100(\alpha^*-1)\%$，$F(\alpha^*)$}\;
\end{algorithm}
