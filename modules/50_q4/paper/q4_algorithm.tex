\begin{algorithm}[H]
\SetAlgoLined
\setstretch{0.92}
\SetAlFnt{\normalsize}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{Input}
\SetKwInput{KwOut}{Output}
\SetKwFor{For}{for}{do}{end for}
\SetKwWhile{While}{while}{do}{end while}
\SetKwIF{If}{ElseIf}{Else}{if}{then}{else if}{else}{end if}
\SetKw{KwRet}{Return}
\caption{Bisection search for minimum heat release multiplier}
\label{alg:q4-bisection}
\KwIn{Baseline time $t_3^*$, weight limit $t_W^{(4)}$, tolerance $\varepsilon_\alpha$, latent heat $L_{\rm pcm}(v)$ for each scan rate}
\KwOut{Minimum multiplier $\alpha^*$, increase ratio $r_{\min}=100(\alpha^*-1)\%$}

\If{$t_W^{(4)}<t_3^*$}{
  \KwRet{weight limit insufficient, problem infeasible}\;
}
$\alpha_L\leftarrow 1$\;
$\alpha_U\leftarrow 2$\;
\While{$t_T^{(4)}(\alpha_U)<t_3^*$}{
  $\alpha_L\leftarrow\alpha_U$\;
  $\alpha_U\leftarrow 2\alpha_U$\;
}
\While{$\alpha_U-\alpha_L>\varepsilon_\alpha$}{
  $\alpha_M\leftarrow(\alpha_L+\alpha_U)/2$\;
  Compute $t_T^{(4)}(\alpha_M)$ via the forward solver from Question~1\;
  \If{$t_T^{(4)}(\alpha_M)\ge t_3^*$}{
    $\alpha_U\leftarrow\alpha_M$\;
  }
  \Else{
    $\alpha_L\leftarrow\alpha_M$\;
  }
}
$\alpha^*\leftarrow\alpha_U$\;
$t_{\rm eff}^{(4)}(\alpha^*)\leftarrow\min\{t_T^{(4)}(\alpha^*),t_W^{(4)}\}$\;
\KwRet{$\alpha^*$, $100(\alpha^*-1)\%$, $t_{\rm eff}^{(4)}(\alpha^*)$}\;
\end{algorithm}
