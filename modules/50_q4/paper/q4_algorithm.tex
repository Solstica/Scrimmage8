\begin{algorithm}[H]
\SetAlgoLined
\setstretch{0.92}
\SetAlFnt{\normalsize}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{Input}
\SetKwInput{KwOut}{Output}
\SetKwInput{KwInit}{Initialize}
\SetKwWhile{While}{while}{do}{end while}
\SetKwIF{If}{ElseIf}{Else}{if}{then}{else if}{else}{end if}
\SetKw{KwRet}{Return}
\caption{问题四最小放热倍率二分搜索}
\label{alg:q4-bisection}
\KwIn{场景$(\beta_{\rm DSC},\theta,t_{3,\theta}^*)$，负重上限$t_W^{(4)}$，倍率容差$\varepsilon_\alpha$，潜热$L_{\rm pcm}(\beta_{\rm DSC})$}
\KwInit{$\alpha_L\leftarrow1$，$\alpha_U\leftarrow2$}
\KwOut{最小放热倍率$\alpha^*$、提高比例$r_{\min}=100(\alpha^*-1)\%$及实际持续时间$t_{\rm eff,\theta}^{(4)}(\alpha^*)$}

\If{$t_W^{(4)}<t_{3,\theta}^*$}{
  \KwRet{负重上限不足，当前场景无可行解}\;
}
调用问题一传热正演模型计算$t_{T,\theta}^{(4)}(1)$，并取$t_{\rm eff,\theta}^{(4)}(1)=\min\{t_{T,\theta}^{(4)}(1),t_W^{(4)}\}$\;
\If{$t_{\rm eff,\theta}^{(4)}(1)\ge t_{3,\theta}^*$}{
  \KwRet{$\alpha^*=1$，$r_{\min}=0$，$t_{\rm eff,\theta}^{(4)}(1)$}\;
}
调用问题一传热正演模型计算$t_{\rm eff,\theta}^{(4)}(\alpha_U)$\;
\While{$t_{\rm eff,\theta}^{(4)}(\alpha_U)<t_{3,\theta}^*$}{
  $\alpha_L\leftarrow\alpha_U$\;
  $\alpha_U\leftarrow2\alpha_U$\;
  调用问题一传热正演模型计算$t_{\rm eff,\theta}^{(4)}(\alpha_U)$\;
}
\While{$\alpha_U-\alpha_L>\varepsilon_\alpha$}{
  $\alpha_M\leftarrow(\alpha_L+\alpha_U)/2$\;
  调用问题一传热正演模型计算$t_{T,\theta}^{(4)}(\alpha_M)$，并取$t_{\rm eff,\theta}^{(4)}(\alpha_M)=\min\{t_{T,\theta}^{(4)}(\alpha_M),t_W^{(4)}\}$\;
  \If{$t_{\rm eff,\theta}^{(4)}(\alpha_M)\ge t_{3,\theta}^*$}{
    $\alpha_U\leftarrow\alpha_M$\;
  }
  \Else{
    $\alpha_L\leftarrow\alpha_M$\;
  }
}
$\alpha^*\leftarrow\alpha_U$\;
调用问题一传热正演模型复算$t_{\rm eff,\theta}^{(4)}(\alpha^*)$\;
\KwRet{$\alpha^*$，$100(\alpha^*-1)\%$，$t_{\rm eff,\theta}^{(4)}(\alpha^*)$}\;
\end{algorithm}
