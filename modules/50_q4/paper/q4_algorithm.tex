\begin{algorithm}[H]
\SetAlgoLined
\setstretch{0.92}
\SetAlFnt{\normalsize}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{Input}
\SetKwInput{KwOut}{Output}
\SetKwFor{For}{for}{do}{end for}
\SetKwWhile{While}{while}{do}{end while}
\SetKwIF{If}{ElseIf}{Else}{if}{then}{else if}{else}{end if}
\SetKw{KwRet}{Return}
\caption{Bisection search for the minimum heat release multiplier}
\label{alg:q4-bisection}
\KwIn{target time $t_3^*$, weight limit $t_W^{(4)}$, tolerance $\varepsilon_\alpha$, latent heat $L_{\rm pcm}(\beta_{\rm DSC})$}
\KwOut{minimum multiplier $\alpha^*$, ratio $r_{\min}=100(\alpha^*-1)\%$}

\If{$t_W^{(4)}<t_3^*$}{
  \KwRet{infeasible under the weight limit}\;
}
Set $\alpha_L\leftarrow1$\;
Set $\alpha_U\leftarrow2$\;
\While{$t_T^{(4)}(\alpha_U)<t_3^*$}{
  Set $\alpha_L\leftarrow\alpha_U$\;
  Double $\alpha_U$\;
}
\While{$\alpha_U-\alpha_L>\varepsilon_\alpha$}{
  Compute $\alpha_M\leftarrow(\alpha_L+\alpha_U)/2$\;
  Evaluate $t_T^{(4)}(\alpha_M)$ with the thermal forward model\;
  \If{$t_T^{(4)}(\alpha_M)\ge t_3^*$}{
    Set $\alpha_U\leftarrow\alpha_M$\;
  }
  \Else{
    Set $\alpha_L\leftarrow\alpha_M$\;
  }
}
Set $\alpha^*\leftarrow\alpha_U$\;
Compute $t_{\rm eff}^{(4)}(\alpha^*)\leftarrow\min\{t_T^{(4)}(\alpha^*),t_W^{(4)}\}$\;
\KwRet{$\alpha^*$, $100(\alpha^*-1)\%$, $t_{\rm eff}^{(4)}(\alpha^*)$}\;
\end{algorithm}
