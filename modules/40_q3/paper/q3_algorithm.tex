\begin{algorithm}[H]
\SetAlgoLined
\setstretch{0.92}
\SetAlFnt{\normalsize}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{输入}
\SetKwInput{KwInit}{初始化}
\SetKwInput{KwOut}{输出}
\SetKwFor{For}{for}{do}{end for}
\caption{问题三双阈值离散厚度枚举算法}
\label{alg:q3-enumeration}
\KwIn{DSC扫描速率$v$、放热数据、问题一模型参数}
\KwInit{候选集合$n\in\{0,1,2,3\}$，时间步$0.25\ \mathrm{s}$，模态数$M=40$}
\KwOut{全部候选的$t_{15},t_{10},t_W,t_{\mathrm{eff},15},t_{\mathrm{eff},10}$及两种最优层数}

计算$d_3(n)=0.3(1+n)\ \mathrm{mm}$、服装质量$m_c(n)$和$t_W(n)=20[40-m_c(n)]$\;
\For{每个候选$n\in\{0,1,2,3\}$}{
    按$37\,^{\circ}\mathrm C$恒温人体边界初始化三层温度和不可逆相变进度\;
    采用问题一模态求解器逐步推进，直至人体侧温度首次低于$10\,^{\circ}\mathrm C$\;
    线性插值得到首次达到$15\,^{\circ}\mathrm C$与$10\,^{\circ}\mathrm C$的时间$t_{15}(n),t_{10}(n)$\;
    计算$t_{\mathrm{eff},\theta}(n)=\min\{t_\theta(n),t_W(n)\}$，$\theta\in\{15,10\}$\;
}
分别计算$n_\theta^*=\arg\max_n t_{\mathrm{eff},\theta}(n)$，$\theta\in\{15,10\}$\;
\end{algorithm}
