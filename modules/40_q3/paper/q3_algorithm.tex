\begin{algorithm}[H]
\SetAlgoLined
\setstretch{0.92}
\SetAlFnt{\normalsize}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{Input}
\SetKwInput{KwOut}{Output}
\SetKwInput{KwInit}{Initialize}
\SetKwFor{For}{for}{do}{end for}
\SetKw{KwRet}{Return}
\caption{双阈值离散厚度枚举算法}
\label{alg:q3-enumeration}
\KwIn{$\beta_{\rm DSC}$、放热曲线$(T,p(T))$及问题一传热模型参数}
\KwInit{$\mathcal N=\{0,1,2,3\}$，$\Delta t=0.25\ \mathrm{s}$，$M=40$}
\KwOut{各方案的$t_{15},t_{10},t_W,t_{\rm eff,15},t_{\rm eff,10}$及$n_{15}^*,n_{10}^*$}
\For{each $n\in\mathcal N$}{
  计算$d_3(n)$、服装质量$m_c(n)$及$t_W(n)=20[40-m_c(n)]$\;
  初始化$T_i(x,0)=37\,^\circ\mathrm C$、$\xi=0$，调用问题一正演模型至$T_1(0,t)\le10\,^\circ\mathrm C$\;
  插值得$t_{15}(n),t_{10}(n)$\;
  对$\theta\in\{15,10\}$计算$t_{\rm eff,\theta}(n)=\min\{t_\theta(n),t_W(n)\}$\;
}
对$\theta\in\{15,10\}$取$n_\theta^*=\arg\max_{n\in\mathcal N}t_{\rm eff,\theta}(n)$\;
\KwRet{$n_{15}^*,n_{10}^*,d_3(n_\theta^*),t_{\rm eff,15}^*,t_{\rm eff,10}^*$及候选方案结果}\;
\end{algorithm}
