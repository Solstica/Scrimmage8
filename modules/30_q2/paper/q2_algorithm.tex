\begin{algorithm}[H]
\SetAlgoLined
\setstretch{0.92}
\SetAlFnt{\normalsize}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{Input}
\SetKwInput{KwInit}{Initialize}
\SetKwInput{KwOut}{Output}
\SetKwFor{While}{while}{do}{end while}
\SetKwIF{If}{ElseIf}{Else}{if}{then}{else if}{else}{end if}
\SetKw{KwRet}{Return}
\SetKwFunction{SolveLinear}{SolveLinear}
\SetKwFunction{InterpThresh}{InterpThresh}
\caption{问题二微小时间段拓展分离变量法}
\label{alg:q2-thermal}
\KwIn{$\rho_i,c_i,k_i,d_i$，$A_{\rm DSC},\beta_{\rm DSC}$，$h_b=4.0$，$h_e=18.46\ \mathrm{W/(m^2\cdot K)}$，$T_\infty=-40\,^{\circ}\mathrm C$，$\Delta t=0.25\ \mathrm s$，$M=40$}
\KwInit{计算$L_{\rm pcm}$与$w(T)$并预求边界本征值；置$T_i(x,0)=37\,^{\circ}\mathrm C$，$\xi=0$，$n=0$，$t_{15},t_{10}=\varnothing$}
\KwOut{$t_{15},t_{10}$及$T_i(x,t)$}
\While{$T_1(0,t_n)>10\,^{\circ}\mathrm C$}{
  由$\bar T_2^n$和历史$\xi$更新$c_{2,\rm eff}^n$与$\alpha_2^n$\;
  线性近似两处界面温度，并构造三层本征展开\;
  由热流连续条件解$[\beta_{12}^n,\beta_{23}^n]^\top\leftarrow\SolveLinear(\cdot)$\;
  推进至$t_{n+1}$并按历史最大值更新$\xi$；若$\xi=1$，后续取$c_{2,\rm eff}=c_2$\;
  \If{$t_{15}=\varnothing$ 且 $T_1(0,t_{n+1})\le15\,^{\circ}\mathrm C$}{
    $t_{15}\leftarrow\InterpThresh(T_1(0,t_n),T_1(0,t_{n+1}),t_n,t_{n+1},15)$\;
  }
  \If{$t_{10}=\varnothing$ 且 $T_1(0,t_{n+1})\le10\,^{\circ}\mathrm C$}{
    $t_{10}\leftarrow\InterpThresh(T_1(0,t_n),T_1(0,t_{n+1}),t_n,t_{n+1},10)$\;
  }
  $n\leftarrow n+1$\;
}
\KwRet{$t_{15},t_{10},T_i(x,t)$}\;
\end{algorithm}
